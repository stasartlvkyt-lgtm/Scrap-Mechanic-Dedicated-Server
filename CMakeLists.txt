function(scrapds_archive name filename remote hash)
  if(EXISTS "${SCRAPDS_DOWNLOAD_DIR}/${filename}")
    set(location "${SCRAPDS_DOWNLOAD_DIR}/${filename}")
  else()
    set(location "${remote}")
  endif()
  FetchContent_Declare(${name} URL "${location}" URL_HASH "SHA256=${hash}" DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
endfunction()
